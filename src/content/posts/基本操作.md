1.用户创建，授权(参考mysql)
CREATE USER 'edpusr'@'%' IDENTIFIED BY '123123';

--业务用户
GRANT CREATE, ALTER, DROP, INDEX, CREATE TEMPORARY TABLES ON *.* TO 'edpusr'@'%';
GRANT SELECT,INSERT,UPDATE,DELETE ON *.* TO 'edpusr'@'%';
GRANT ALTER ROUTINE,CREATE ROUTINE,EXECUTE ON *.* TO 'edpusr'@'%';

--dba用户
GRANT ALL ON edpusr.* TO 'edpusr'@'%';

2.审计日志
#设置参数
SET GLOBAL audit_log = 1;   -- 开启审计日志
SET GLOBAL log_output = 'table';  -- 将日志存储到系统表（默认为文件）
SET GLOBAL long_query_time = 2;  -- 单位：秒（例如记录执行超过 2 秒的 SQL，0 表示记录所有 SQL）

#创建策略
CREATE AUDIT POLICY audit_slow (
  Sql_commands = 'SELECT,INSERT',  -- 指定记录的操作类型
  Enable = 'Y'
);

#查看慢 SQL（执行超过 2 秒）
SELECT start_time,
       user_host,
	   user,db,
	   query_time,
	   sql_text
FROM gbase.audit_log
WHERE 1=1
 AND sql_text like 'sql内容'
 AND query_time > '00:00:10';
