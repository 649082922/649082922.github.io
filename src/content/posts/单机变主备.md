####################################修改参数####################################

1.修改dm.ini
vi /dm8_data/rural/dm.ini
INSTANCE_NAME                   = rural_01    #数据库实例名                          ?!
ALTER_MODE_STATUS               = 0           #不允许手工方式修改实力模式/状态/OGUID ?
ENABLE_OFFLINE_TS               = 2           #不允许备库OFFLINE表空间               ?
MAL_INI                         = 1           #打开 MAL 系统                         ?
ARCH_INI                        = 1           #打开归档配置                          ?
SVR_LOG                         = 1           #开启异步SQL日志                       ?

2.新增dmarch.ini
vi /dm8_data/rural/dmarch.ini
ARCH_WAIT_APPLY            = 0            #0：高性能   1：事务一致
[ARCHIVE_LOCAL]
  ARCH_TYPE                = LOCAL        #本地归档类型
  ARCH_DEST                = /dm8_arch    #本地归档存放路径
  ARCH_FILE_SIZE           = 1024         #单个归档大小，单位 MB
  ARCH_SPACE_LIMIT         = 51200        #归档上限，单位 MB

[ARCHIVE_REALTIME]
  ARCH_TYPE                = REALTIME     #实时归档类型
  ARCH_DEST                = rural_02     #实时归档目标实例名

3.新增dmmal.ini
vi /dm8_data/rural/dmmal.ini

MAL_CHECK_INTERVAL       = 10                  #MAL 链路检测时间间隔
MAL_CONN_FAIL_INTERVAL   = 10                  #判定 MAL 链路断开的时间
MAL_TEMP_PATH            = /dm8_data/malpath/  #临时文件目录
MAL_BUF_SIZE             = 512                 #单个 MAL 缓存大小，单位 MB
MAL_SYS_BUF_SIZE         = 2048                #MAL 总大小限制，单位 MB
MAL_COMPRESS_LEVEL       = 0                   #MAL 消息压缩等级，0 表示不压缩

[MAL_INST1]
MAL_INST_NAME            = rural_01       #实例名，和 dm.ini 的 INSTANCE_NAME 一致
MAL_HOST                 = 192.168.81.81  #MAL 系统监听 TCP 连接的 IP 地址
MAL_PORT                 = 5336           #MAL 系统监听 TCP 连接的端口
MAL_INST_HOST            = 192.168.56.81  #实例的对外服务 IP 地址
MAL_INST_PORT            = 5236           #实例对外服务端口，和 dm.ini 的 PORT_NUM 一致
MAL_DW_PORT              = 5436           #实例对应的守护进程监听 TCP 连接的端口
MAL_INST_DW_PORT         = 5536           #实例监听守护进程 TCP 连接的端口

[MAL_INST2]
MAL_INST_NAME            = rural_02
MAL_HOST                 = 192.168.81.82
MAL_PORT                 = 5336
MAL_INST_HOST            = 192.168.56.82
MAL_INST_PORT            = 5236
MAL_DW_PORT              = 5436
MAL_INST_DW_PORT         = 5536

4.新增dmwatcher.ini
vi /dm8_data/rural/dmwatcher.ini

[GRP1]
  DW_TYPE                  = GLOBAL  #全局守护类型
  DW_MODE                  = AUTO  #MANUAL：故障手切 AUTO：故障自切
  DW_ERROR_TIME            = 20  #远程守护进程故障认定时间
  INST_ERROR_TIME          = 20  #本地实例故障认定时间
  INST_RECOVER_TIME        = 60  #主库守护进程启动恢复的间隔时间
  INST_OGUID               = 45331  #守护系统唯一 OGUID 值
  INST_INI                 = /dm8_data/rural/dm.ini  #dm.ini 文件路径
  INST_AUTO_RESTART        = 1  #打开实例的自动启动功能
  INST_STARTUP_CMD         = /dm8/bin/dmserver  #命令行方式启动
  RLOG_SEND_THRESHOLD      = 0  #指定主库发送日志到备库的时间阈值，默认关闭
  RLOG_APPLY_THRESHOLD     = 0  #指定备库重演日志的时间阈值，默认关闭

5.新增dmmonitor_rural_P.ini(只有两台就配备库)
vi /dm8/bin/dmmonitor_rural_P.ini
MON_DW_CONFIRM             = 1                #0：非确认（故障手切） 1：确认（故障自切）
MON_LOG_PATH               = /dm8/log/rural/  #监视器日志文件存放路径?
MON_LOG_INTERVAL           = 60               #每隔 60s 定时记录系统信息到日志文件
MON_LOG_FILE_SIZE          = 512              #单个日志大小，单位 MB
MON_LOG_SPACE_LIMIT        = 2048             #日志上限，单位 MB

################单监视器不用配置这6个参数################
MON_INST_NUM = 2        #监视器系统中的实例总数
MON_HB_INTERVAL = 60    #各实例通信的心跳校验间隔
MON_BRO_INTERVAL = 100  #各实例通信的心跳间隔
MON_VOTE_INTERVAL = 100 #各实例通信的基础投票间隔
MON_ID =1               #注意修改P=1，S=2
MON_MIN =45633          #使用与现有其他确认监视器端口不同的MID（递增），登记入数数据库信息表

[GRP1]
MON_INST_OGUID           = 45331       #组 GRP1 的唯一 OGUID 值
#以下配置为监视器到组GRP1的守护进程的连接信息,以一IP:POST||的形式配置
#IP,PORT 对应dmmal.ini中的 MAL_HOST , MAL_DW_PORT
MON_DW_IP                = 192.168.81.81:5436  #IP 对应 MAL_HOST，PORT 对应 MAL_DW_PORT
MON_DW_IP                = 192.168.81.82:5436

####################################备份恢复####################################

1.关库备份
/dm8/bin/DmService_rural_01 stop

2.备份
cd /dm8/bin
./dmrman
backup database '/dm8_data/rural/dm.ini' backupset '/backup/backup_file';

3.传输scp
scp -r /backup/backup_file dmdba@192.168.81.82:/backup

4.恢复
/dm8/bin/dmrman CTLSTMT="RESTORE DATABASE TO '/dm8_data/rural' FROM BACKUPSET '/backup/backup_file'"
/dm8/bin/dmrman CTLSTMT="RECOVER DATABASE '/dm8_data/rural/dm.ini' FROM BACKUPSET '/backup/backup_file'"
/dm8/bin/dmrman CTLSTMT="RECOVER DATABASE '/dm8_data/rural/dm.ini' UPDATE DB_MAGIC"

####################################注册服务####################################

1.注册dmservice服务
#主库注册
/dm8/script/root/dm_service_installer.sh -t dmserver -dm_ini /dm8_data/rural/dm.ini -p _rural_01

#备库注册
/dm8/script/root/dm_service_installer.sh -t dmserver -dm_ini /dm8_data/rural/dm.ini -p _rural_02

#回退
/dm8/script/root/dm_service_uninstaller.sh -n DmService_rural_01

2.注册dmwatcher服务
#主库注册
/dm8/script/root/dm_service_installer.sh -t dmwatcher -p _rural_01 -watcher_ini /dm8_data/rural/dmwatcher.ini

#备库注册
/dm8/script/root/dm_service_installer.sh -t dmwatcher -p _rural_02 -watcher_ini /dm8_data/rural/dmwatcher.ini

备注:如果配错了,使用删除自启
/dm8/script/root/dm_service_uninstaller.sh -n DmWatcherService_rural_01

3.注册monitor服务
JK-1机：
/dm8/script/root/dm_service_installer.sh -t dmmonitor -p \
DMMONITOR_rural_P -monitor_ini /dm8/bin/dmmonitor_rural_P.ini

9.修改启动模式为mount(AB机器)
vim /dm8/bin/DmService_rural_01
vim /dm8/bin/DmService_rural_02

#mount or open
START_MODE=mount

10.启动数据库
启动数据库,执行命令(AB机器)
--主库
/dm8/bin/DmService_rural_01 start
/dm8/bin/disql SYSDBA/SYSDBA@192.168.81.81:5236
select STATUS$  from v$instance;
sp_set_para_value(1,'alter_mode_status',1);
sp_set_oguid(45331);
alter database primary;
sp_set_para_value(1,'alter_mode_status',0);

--备库
/dm8/bin/DmService_rural_02 start
/dm8/bin/disql SYSDBA/SYSDBA@192.168.81.82:5236
select STATUS$  from v$instance;
sp_set_para_value(1,'alter_mode_status',1);
sp_set_oguid(45331);
alter database standby;
sp_set_para_value(1,'alter_mode_status',0);

11.启动守护进程
/dm8/bin/DmWatcherService_rural_01 start
/dm8/bin/DmWatcherService_rural_02 start

确保两个节点全部为open再进行下一步

12.启动监视器
/dm8/bin/DmMonitorServiceDMMONITOR_rural_P start
