2.dm.ini参数调优
参考:https://eco.dameng.com/document/dm/zh-cn/pm/physical-storage
使用sysdba用户登陆dm管理工具或disql工具,执行以下脚本
如果在disql命令行中执行,需要先执行 set serverout on
然后再执行脚本,将执行后显示的 sp_set_para_value 部分的sql粘贴出来,重新再执行

###特别注意,小心修改这两个参数 RECYCLE,BUFFER

disql SYSDBA/SYSDBA@192.168.81.81:5236

sp_set_para_value(2,'MEMORY_POOL',         8000        ); --内存池大小(mb)
sp_set_para_value(2,'MEMORY_N_POOLS',      59          ); --内存池数量
sp_set_para_value(2,'MEMORY_TARGET',       30000       ); --内存共享池目标大小（MB）
sp_set_para_value(2,'BUFFER',              320000      ); --初始系统缓冲区大小（MB）
sp_set_para_value(2,'BUFFER_POOLS',        301         ); --缓冲池数量
sp_set_para_value(2,'RECYCLE',             32000       ); --系统回收缓冲区大小（MB）
sp_set_para_value(2,'RECYCLE_POOLS',       51          ); --回收缓冲池的数量

sp_set_para_value(2,'FAST_ROLL_PAGES',     9999999     ); --快速滚动页面的页数
sp_set_para_value(2,'MEMORY_MAGIC_CHECK',  1           ); --是否开启对所有内存池的校验,0关,1开,2加强开
sp_set_para_value(2,'ENABLE_FREQROOTS',    1           ); --是否收集频繁使用的页面

4.创建表空间及备份
说明:创建rural用户,密码为:admin1234,数据存在rural表空间中,权限给 PUBLIC/RESOURCE/VTI/SOI/SVI
创建表空间时:若/dm8_data分区容量小于500G,则表空间总大小不超/dm8_data分区容量的70%;
             若/dm8_data分区容量大于500G,则表空间文件上线可设置为500G;

disql SYSDBA/SYSDBA@192.168.81.81:5236

--表空间和数据库用户应用自己建
--创建表空间
create tablespace "rural" datafile '/dm8_data/rural/rural.DBF' size 10240 AUTOEXTEND ON NEXT 2048;
--创建索引表空间(数据量小项目可以不必,数据量大的项目需要分开维护)
create tablespace "rural_idx" datafile '/dm8_data/rural/rural_idx.DBF' size 5120 AUTOEXTEND ON NEXT 2048;
--创建用户
CREATE USER "rural" IDENTIFIED BY "admin1234" DEFAULT TABLESPACE "rural" DEFAULT INDEX TABLESPACE "rural_idx";
--赋予权限
GRANT "PUBLIC","RESOURCE","VTI","SOI","SVI" TO "rural";
--取消用户登陆限制
alter user "rural" limit failed_login_attemps unlimited,password_lock_time unlimited,password_grace_time unlimited;

--删除用户
drop user "rural" cascade;
--删除表空间
drop tablespace "rural";
drop tablespace "rural_idx";

--创建只读用户,DBQUERY
CREATE USER "DBQUERY" IDENTIFIED BY "DBQUERY168" read only;
--赋予权限
grant CREATE SESSION,SELECT ANY TABLE,SELECT ANY VIEW TO "DBQUERY";
--更改sysdba密码
alter user "SYSDBA" identified by "oracle";

4>归档自动删除
注意：归档之前已经配置，不需要再次执行
###按照归档大小删除
ARCH_SPACE_LIMIT：Redo日志归档空间限制，当同一节点的本地归档文件或远程归档文件达到限制值时，
系统自动删除最早生成的归档日志文件。0表示无空间限制，取值范围（1024M~2147483647M），
对本地归档和远程归档有效，缺省为0。可手动修改dmarch.ini中ARCH_SPACE_LIMIT参数，也可执行sql:
alter database modify archivelog 'DEST=/dm8/arch, TYPE=LOCAL, FILE_SIZE=200, SPACE_LIMIT=10240';

###按照时间删除
ARCH_RESERVE_TIME：归档日志保留时间，单位分钟，取值范围（0~4294967294）。
只对远程归档和本地归档有效，服务器每隔5分钟检查是否存在超过保留时间的归档并删除。
默认为0，表示不删除归档。手动修改dmarch.ini中ARCH_RESERVE_TIME参数，更改为1440即归档保留24小时。

2.定时全库收集统计信息(可选)
--创建统计信息记录表
drop table MY_TAB_STATS;
create table MY_TAB_STATS(tab_num int identity(1,1),
                          tab_name varchar2(100),
						  stat_time timestamp(6),
						  stat_err varchar(1000));
--初始化统计信息记录表
truncate table MY_TAB_STATS;
select * from MY_TAB_STATS;
insert into MY_TAB_STATS(TAB_NAME) select table_name from all_tables where owner='EKP';
update MY_TAB_STATS set stat_time = to_date('1970-01-01 00:00:00');
commit;

--开始更新
sf_set_session_para_value('HAGR_HASH_SIZE',100000000);
set serveroutput on
call stat_zzj(0,100);
create or replace procedure stat_zzj(a int,b int)
as
num_start int;
num_end int;
BEGIN
  num_start=a;
  num_end=b;
  FOR TB IN ( select tab_num,tab_name from MY_TAB_STATS where tab_num > a and tab_num <b ) loop
    begin
	  DBMS_STATS.GATHER_TABLE_STATS('EKP',TB.TAB_NAME,NULL,100,TRUE,'FOR ALL COLUMNS SIZE AUTO');
      UPDATE MY_TAB_STATS SET STAT_TIME = SYSDATE WHERE TAB_NUM = TB.TAB_NUM;
	  COMMIT;
	  print TB.TAB_NAME||' Done,Current Time is '|| sysdate;
	exception
	  when others then
	  update MY_TAB_STATS SET STAT_TIME = SYSDATE,
                              STAT_ERR  = SQLCODE||''||SQLERRM
					    WHERE TAB_NUM = TB.TAB_NUM;
	END;
  END LOOP;
END;

--创建定时收集统计信息的作业
call SP_CREATE_JOB('stat_zzj',1,0,'',0,0,'',0,'');
call SP_JOB_CONFIG_START('stat_zzj');
call SP_JOB_SET_EP_SEQNO('stat_zzj',0);
call SP_ADD_JOB_STEP('stat_zzj', 'step1', 0, 'call stat_zzj(0,100);', 0, 0, 0, 0, NULL, 0);
call SP_ADD_JOB_SCHEDULE('stat_zzj', 'step2', 1, 0, 0, 0, 0, NULL, NULL, '2023-05-31 22:00:00', NULL, '');
call SP_JOB_CONFIG_COMMIT('stat_zzj');
