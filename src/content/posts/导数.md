1.查询两端必需相同项
SELECT '字符集',CASE SF_GET_UNICODE_FLAG() WHEN '0' THEN 'GBK18030' WHEN '1' then 'UTF-8' when '2' then 'EUC-KR' end
union all
SELECT '页大小',cast(PAGE()/1024 as varchar)
union all
SELECT '簇大小',cast(SF_GET_EXTENT_SIZE() as varchar)
union all
SELECT '大小写敏感',cast(SF_GET_CASE_SENSITIVE_FLAG() as varchar)
union ALL
select '是否以字符为长度',case SF_GET_LENGTH_IN_CHAR() when '0' then '否' when '1' then '是' end
union all
select '补空格', PARA_VALUE from v$DM_INI WHERE PARA_NAME = 'BLANK_PAD_MODE';

2.导入导出
# 用户级导出（dexp）
nohup ./dexp USERID="SYSDBA"/"Dm08serv$"@LOCALHOST:5236  \
  FILE=full_20260414_%U.dmp \
  LOG=full_20260414_exp.log \
  DIRECTORY=/dm8_data/backup/backup_01 \
  OWNER=YBTS \
  PARALLEL=32 \
  TABLE_PARALLEL=8 \
  COMPRESS=Y \
#2024 年后dm8 新参数  COMPRESS_LEVEL=9 \
  LOG_WRITE=Y \
  FEEDBACK=10000 \
  > /dev/null 2>&1 &

# 用户级导入（dimp）

nohup ./dimp USERID=SYSDBA/"Dm08serv$"@LOCALHOST:5236 \
  FILE=full_20260414_01.dmp \
  LOG=full_20260414_imp.log \
  DIRECTORY=/dm8_data/backup/backup_01 \
  OWNER=YBTS \
  PARALLEL=32 \
  TABLE_PARALLEL=8 \
  IGNORE=Y \
  FAST_LOAD=Y \
  COMMIT_ROWS=10000 \
  LOG_WRITE=Y \
  > /dev/null 2>&1 &

Oracle expdp/impdp	           达梦 dexp/dimp	                    说明
"/ as sysdba"	               SYSDBA/密码@IP:端口	                达梦不支持操作系统认证的简写，需显式写连接串
directory=dump_dir	           DIRECTORY=/dmbackup/dexp	            dexp 直接用路径；dexpdp 需先 CREATE DIRECTORY
dumpfile=xxx.dump	           FILE=xxx.dmp	                        达梦用 .dmp 后缀，支持 %U 自动扩展
logfile=xxx.log	               LOG=xxx.log	                        日志文件
schemas=xxx	                   OWNER=xxx 或 SCHEMAS=xxx	            用户级用 OWNER，模式级用 SCHEMAS
parallel=32	                   PARALLEL=32	                        达梦范围 1~100，默认 16
compression=all	               COMPRESS=Y COMPRESS_LEVEL=9	        达梦压缩等级 0~9，默认 1
cluster=n	                   无直接对应	                        达梦 MPP 环境用 #{mpp_type=local}
TRANSFORM=SEGMENT_CREATION:n   无直接对应	                        达梦没有延迟段创建概念，无需此参数
TABLE_EXISTS_ACTION=APPEND	   IGNORE=Y 或 TABLE_EXISTS_ACTION=APPEND	dimp 两者都支持
无	                           FAST_LOAD=Y	                        dimp 专用，使用 dmfldr 快速装载，大幅提升导入性能
无	                           TABLE_PARALLEL=8	                    每张表使用的子线程数(1~50)
无	                           COMMIT_ROWS=10000	                批量提交行数，配合 FAST_LOAD 使用
